59 function isConfirmedUnavailable(live) {
60   if (!live || !live.status) return false;
61   const status = String(live.status).toLowerCase().trim().replace(/[\s-]+/g, '_');
62   // Network/HTTP failures are not proof of a permanently unavailable product.
63   if (status === 'error' || status === 'timeout' || status === 'blocked') return false;
64   if (['out_of_stock', 'sold_out', 'discontinued', 'permanently_unavailable'].includes(status)) return true;
65   const availability = String(live.availability || '').toLowerCase();
66   if (/discontinued|sold\s*out|out\s*of\s*stock|no\s*longer\s*available/.test(availability)) return true;
67   // An ambiguous N/A, unavailable or missing listing must persist across TWO fresh checks.
68   return (['not_available', 'unavailable', 'n/a', 'na', 'not_found'].includes(status)
69     || /(^n\/a$|not\s*available|unavailable)/.test(availability))
70     && (Number(live.unavailable_checks) || 0) >= 2;

73 function toLive(result, prev = {}) {

75   const s = String(result.status || '').toLowerCase().trim().replace(/[\s-]+/g, '_');
76   const availabilityLabel = String(result.availability || '').toLowerCase();
77   const unavailable = ['out_of_stock', 'sold_out', 'discontinued', 'permanently_unavailable',
78     'not_available', 'unavailable', 'n/a', 'na', 'not_found'].includes(s)
